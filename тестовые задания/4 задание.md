структура проекта:

получение информации о пользователе через GetUserInfo  GET /api/user?userId=ID

Цепочка атаки на `/api/user`:
1. Получаем всех пользователей  `/api/user?userId=` из за ошибки обработки, выбираем подходящую цель
2. Получаем валидный reset токен для своего email `POST /api/user/password/request + body json с email` 
3. Сбрасываем пароль нужного пользователя зная его почту из пункта 1, используем наш валидный токен  `POST /api/user/password/reset + body {"email": "admin@app.com","token": "ATTACKER_VALID_RESET_TOKEN","newPassword": "Password123"}`

```C#
[HttpPost("password/reset")]
public IActionResult Reset([FromBody] ResetPasswordModel req)
{
	//здесь я попробовал через ORM запрос для разнообразия чтобы не писать один SQL, теперь одновременно проверяется и юзер и принадлежность токена
    var account = _db.Users.FirstOrDefault(x =>
        x.Email == req.Email &&
        x.ResetToken == req.Token &&
        x.ResetTokenTTL > DateTime.UtcNow);

    if (account == null)
    {
        return BadRequest("Invalid or expired token");
    }

    byte[] salt = RandomNumberGenerator.GetBytes(16);
    using var pbkdf2 = new Rfc2898DeriveBytes(
        req.NewPassword,
        salt,
        100000,
        HashAlgorithmName.SHA256);

    byte[] hash = pbkdf2.GetBytes(32);
    account.PasswordHash = $"{Convert.ToBase64String(salt)}:{Convert.ToBase64String(hash)}";

    account.ResetToken = null;
    account.ResetTokenTTL = null;

    _db.SaveChanges();

    return Ok("Password updated");
}
```

Конкретное описание уязвимости:
LFI в GeneratePdfReportControlller, из за того что пользовательский html открывается через file://, а chromium запускается с флагом --allow-file-access-from-files, а самописная санитизация написано плохо и ее легко обойти за счет например использования localhost/path_to_read_file
SSTI в emailController, потому что пользовательский ввод попадает в шаблон, где опять же самописная санитизация, которая не уберет `@(2*2)` или функцию разбитую по строкам "sys"+"tem"
Максимальный реальный риск:
Полная компрометация сервера
Шаги по эксплуатации, полезная нагрузка для реализации этого риска и пояснения к ней:  
1. Любой авторизированный пользователь обращается к `/PdfGenerator`, используя payload 
```
<html>
<body>
<img src="x" onerror="
    var x = new XMLHttpRequest();
    x.open('GET', 'file://localhost/etc/app/secrets/admin_api_key.txt', false);
    x.send();
    document.body.innerText = x.responseText;">
</body>
</html>
```
Читаем API ключ администратора, payload сработает потому что тег `<img>` не удаляет текущий фильтр, а onerror нет в запрещенных events
2. После получения ключа через lfi воспользуемся SSTI в Razor, потому что req.Comment конкатенируется без какой либо фильтрации, далее при рендеринге происходит слабая проверка по запрещенным словам, поэтому можно использовать какой то такой payload в комментарии:
```
@(7*7)
```
Тело запроса будет выглядеть следующим образом: 
```
{  
"email": "aboba@app.com",  
"adminApiKey": "adminApiKey",  
"comment": "@(7*7)"
}
```
После чего получим 49 в комментарии, что докажет SSTI, и в зависимости от цели можем использовать RCE на свое усмотрение
**Рекомендации по устранению, необходимые замены в коде и комментарии, каким образом они обеспечат защиту**
Я как то читал что лучше использовать специализированные библиотеки нежели писать проверки самому, поэтому я добавлю библиотеку htmlSanitazer для .Net и изменю логику в на белые лист
```C# 
using Microsoft.AspNetCore.Authorization;  
using Microsoft.AspNetCore.Mvc;  
using PuppeteerSharp;
using Ganss.Xss;  
  
[ApiController]  
[Route("[controller]")]  
[Authorize(Roles = "user")]  
public class PdfGeneratorController : ControllerBase  
{  
    // создаем один санитайзер  
    private static readonly HtmlSanitizer PdfHtmlSanitizer = CreatePdfHtmlSanitizer();  
  
    private static HtmlSanitizer CreatePdfHtmlSanitizer()  
    {  
        var sanitizer = new HtmlSanitizer();  
  
        // отчистим предлагаемые библиотекой списки и потом добавим те что нужны для нашей задачи  
        sanitizer.AllowedTags.Clear();  
        sanitizer.AllowedAttributes.Clear();  
        sanitizer.AllowedCssProperties.Clear();  
        sanitizer.AllowedSchemes.Clear();  
        sanitizer.UriAttributes.Clear();  
  
        // разрешаем только самые простые теги  
        foreach (var tag in new[]  
        {  
            "div", "span", "p", "br", "hr",  
            "h1", "h2", "h3", "h4",  
            "strong", "em", "b", "i", "u", "small",  
            "table", "thead", "tbody", "tr", "th", "td",  
            "ul", "ol", "li"  
        })  
        {  
            sanitizer.AllowedTags.Add(tag);  
        }  
  
        // разрешаем безопасные атрибуты  
        foreach (var attr in new[]  
        {  
            "class", "style", "colspan", "rowspan", "align"  
        })  
        {  
            sanitizer.AllowedAttributes.Add(attr);  
        }  
  
        // разрешаем только простые свойства для CSS, чтобы в них нельзя было спрятать сложную логику  
        foreach (var css in new[]  
        {  
            "color", "background-color",  
            "font", "font-family", "font-size", "font-weight", "font-style",  
            "text-align", "text-decoration",  
            "border", "border-collapse", "border-spacing",  
            "width", "height", "max-width",  
            "margin", "margin-left", "margin-right", "margin-top", "margin-bottom",  
            "padding", "padding-left", "padding-right", "padding-top", "padding-bottom",  
            "display", "vertical-align", "white-space"  
        })  
        {  
            sanitizer.AllowedCssProperties.Add(css);  
        }  
  
        return sanitizer;  
    }  
  
    [HttpPost] 
    public async Task<IActionResult> GeneratePdf()  
    {  
        using var reader = new StreamReader(Request.Body);  
        var html = await reader.ReadToEndAsync();  
  
        //если html пустой то отсылаем обратно  
        if (string.IsNullOrWhiteSpace(html))  
        {  
            return BadRequest(new { error = "HTML body is required" });  
        }  
  
        //применяем наш фильтр для html  
        var safeHtml = PdfHtmlSanitizer.Sanitize(html);  
  
        //проверяем что осталось от html, если ничего то возвращаем ошибку  
        if (string.IsNullOrWhiteSpace(safeHtml))  
        {  
            return BadRequest(new { error = "HTML body does not contain allowed content" });  
        }  
  
        await new BrowserFetcher().DownloadAsync();  
  
        await using var browser = await Puppeteer.LaunchAsync(new LaunchOptions  
        {  
            Headless = true,  
            //убираем --allow-file-access-from-files, чтобы не давать читать локальные файлы  
            Args = new[]  
            {  
                "--no-sandbox",  
            }  
        });  
  
        await using var page = await browser.NewPageAsync();  
  
        // как доп защиту используем отключение javascript  
        await page.SetJavaScriptEnabledAsync(false);  
  
        //теперь не создаем временный файл, а просто храним его в памяти  
        await page.SetContentAsync(safeHtml, new NavigationOptions  
        {  
            WaitUntil = new[] { WaitUntilNavigation.Load },  
            Timeout = 10_000  
        });  
          
        var pdfBytes = await page.PdfDataAsync(new PdfOptions  
        {  
            PrintBackground = true  
        });  
  
        return File(pdfBytes, "application/pdf", "report.pdf");  
    }  
}
```

```c#
using RazorLight;  
  
public class TemplateService  
{  
    private readonly RazorLightEngine _engine;  
  
    public TemplateService()  
    {  
        _engine = new RazorLightEngineBuilder()  
            .UseMemoryCachingProvider()  
            .Build();  
    }  
  
    public async Task<string> RenderTrusted(string template, object model)  
    {  
        if (string.IsNullOrWhiteSpace(template))  
        {  
            throw new ArgumentException("Template must not be empty", nameof(template));  
        }  
  
        string key = Guid.NewGuid().ToString();  
  
        return await _engine.CompileRenderStringAsync(  
            key,  
            template,  
            model  
        );  
    }  
}
```
