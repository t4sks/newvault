[HttpPost("password/reset")]
public IActionResult Reset([FromBody] ResetPasswordModel req)
{
	//здесь я попробовал через ORM запрос для разнообразия чтобы не писать один SQL, теперь одновременно проверяется и юзер и принадлежность токена
    var account = _db.Users.FirstOrDefault(x =>
        x.Email == req.Email &&
        x.ResetToken == req.Token &&
        x.ResetTokenTTL > DateTime.UtcNow);

    if (account == null)
    {
        return BadRequest("Invalid or expired token");
    }

    byte[] salt = RandomNumberGenerator.GetBytes(16);
    using var pbkdf2 = new Rfc2898DeriveBytes(
        req.NewPassword,
        salt,
        100000,
        HashAlgorithmName.SHA256);

    byte[] hash = pbkdf2.GetBytes(32);
    account.PasswordHash = $"{Convert.ToBase64String(salt)}:{Convert.ToBase64String(hash)}";

    account.ResetToken = null;
    account.ResetTokenTTL = null;

    _db.SaveChanges();

    return Ok("Password updated");
}

    //Написано убого, проще использовать готовые библиотеки для санитизации потому что там уже все продумано  
    private static string SanitizeHtml(string html)  
    {  
        if (string.IsNullOrWhiteSpace(html))  
            return "<html><body></body></html>";  
        var sanitized = html;  
        sanitized = Regex.Replace(  
            sanitized,  
            @"<script\b[^>]*>[\s\S]*?</script>",  
            string.Empty,  
            RegexOptions.IgnoreCase | RegexOptions.CultureInvariant);  
        sanitized = Regex.Replace(  
            sanitized,  
            @"</?(iframe|object|embed|frame|frameset|meta|link)\b[^>]*>",  
            string.Empty,  
            RegexOptions.IgnoreCase | RegexOptions.CultureInvariant);  
        var eventAttributes = LoadEventAttributes();  
        foreach (var attr in eventAttributes)  
        {  
            var escapedAttr = Regex.Escape(attr);  
  
            sanitized = Regex.Replace(  
                sanitized,  
                $@"\s{escapedAttr}\s*=\s*(""[^""]*""|'[^']*'|[^\s>]+)",  
                string.Empty,  
                RegexOptions.IgnoreCase | RegexOptions.CultureInvariant);  
        }  
        sanitized = Regex.Replace(  
            sanitized,  
            @"file:///",  
            string.Empty,  
            RegexOptions.IgnoreCase | RegexOptions.CultureInvariant);  
        return sanitized;  
    }  
  
    private static HashSet<string> LoadEventAttributes()  
    {  
        var path = Path.Combine(Directory.GetCurrentDirectory(), EventsListFile);  
        if (!System.IO.File.Exists(path))  
            return new HashSet<string>();  
        return System.IO.File.ReadAllLines(path)  
            .Select(x => x.Trim())  
            .Where(x => !string.IsNullOrWhiteSpace(x))  
            .ToHashSet(StringComparer.OrdinalIgnoreCase);  
    }
