3. Хранение пароля в кодировке base64, 
место в коде:
```
 password_b64 = base64.b64encode(password.encode()).decode()

    db = get_db()
    db.execute(
        f"INSERT INTO users(email, password) VALUES ('{email}', '{password_b64}')" #тут еще и скуля^-^
```

Объяснить почему это проблема:
При хранении паролей пользователей в base64, мы получаем риск что при утечке бд, все учетки пользователей будут скомпрометированы, поэтому необходимо хранить хэши, чтобы даже при утечке перебор значений занял достаточно много времени или вообще был невозможен + необходимо так же добавить соль, я бы исправил примерно так через bcrypt потому что он не требует сложной настройки как аргон
```
salt = bcrypt.gensalt()

password_hash = bcrypt.hashpw(password.encode(), salt)
```

при дебаг режиме фласк позволяет исполнять питон код в браузере при любой ошибке
Исправление: 
```
app.run()
```

```python
import base64
import logging
import sqlite3
from flask import Flask, request, jsonify, session

app = Flask(__name__)
app.config["SECRET_KEY"] = "dev-secret-key"

DB = "app.db"
ADMIN_TOKEN = "admin_123456_secret"

logging.basicConfig(level=logging.INFO)

def get_db():
    return sqlite3.connect(DB)

@app.route("/register", methods=["POST"])
def register():
    email = request.form.get("email")
    password = request.form.get("password")

    password_b64 = base64.b64encode(password.encode()).decode()

    db = get_db()
    db.execute(
        f"INSERT INTO users(email, password) VALUES ('{email}', '{password_b64}')"
    )
    db.commit()
    db.close()

    logging.info(f"Registered user: email={email}, password={password}")
    return jsonify({"status": "ok"})

@app.route("/login", methods=["POST"])
def login():
    email = request.form.get("email")
    password = request.form.get("password")
    password_b64 = base64.b64encode(password.encode()).decode()

    db = get_db()
    user = db.execute(
        f"SELECT id, email FROM users WHERE email='{email}' AND password='{password_b64}'"
    ).fetchone()
    db.close()

    if user is None:
        return jsonify({"error": "Wrong email or password"}), 401

    session["user_id"] = user[0]
    session["email"] = user[1]

    return jsonify({"status": "logged_in"})

@app.route("/notes", methods=["GET"])
def notes():
    if not session.get("user_id"):
        return jsonify({"error": "Unauthorized"}), 401

    user_id = request.args.get("user_id")

    db = get_db()
    rows = db.execute(
        f"SELECT id, title, body FROM notes WHERE user_id = {user_id}"
    ).fetchall()
    db.close()

    return jsonify([
        {"id": row[0], "title": row[1], "body": row[2]}
        for row in rows
    ])

@app.route("/admin/users", methods=["GET"])
def admin_users():
    token = request.headers.get("X-Admin-Token")

    if token != ADMIN_TOKEN:
        return jsonify({"error": "Forbidden"}), 403

    db = get_db()
    users = db.execute("SELECT id, email, password FROM users").fetchall()
    db.close()

    return jsonify([
        {"id": user[0], "email": user[1], "password": user[2]}
        for user in users
    ])

if __name__ == "__main__":
    app.run(debug=True)
```
