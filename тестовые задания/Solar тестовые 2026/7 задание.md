```python
from pathlib import Path
from impacket.dpapi import DPAPI_BLOB


MACHINE_KEY_FILE = Path(
    "ProgramData/Microsoft/Crypto/RSA/MachineKeys/"
    "fc58e716dd94cece30c087d4db914620_e1d345bd-ebc1-4d50-8b78-3f3e787e0953"
)

MASTERKEY_HEX = (
    "43d0f1e158b63d8e5e992cb886a142bdce9f7c0404417f39699e7f53e73183f7"
    "bbf090a918bde8cb1c9821dac4094f5a2c2ad18e0642d4c0667a7f5c9b2d81a7"
)


def calc_dpapi_blob_len(blob):
    return (
        4 + 16 + 4 + 16 + 4 +
        4 + blob["DescriptionLen"] +
        4 + 4 +
        4 + blob["SaltLen"] +
        4 + blob["HMacKeyLen"] +
        4 + 4 +
        4 + len(blob["HMac"]) +
        4 + blob["DataLen"] +
        4 + blob["SignLen"]
    )


data = MACHINE_KEY_FILE.read_bytes()


marker = bytes.fromhex("01000000d08c9ddf")

offset = data.find(marker)

if offset == -1:
    raise SystemExit("[-] DPAPI blob not found")

print("[+] DPAPI blob offset:", offset)

blob = DPAPI_BLOB(data[offset:])

blob_len = calc_dpapi_blob_len(blob)
print("[+] DPAPI blob length:", blob_len)

blob = DPAPI_BLOB(data[offset:offset + blob_len])

masterkey = bytes.fromhex(MASTERKEY_HEX)

clear = blob.decrypt(masterkey)

if clear is None:
    raise SystemExit("[-] DPAPI decrypt failed")

print("[+] DPAPI blob decrypted")
print("[+] First bytes:", clear[:16].hex(" "))

Path("rsa2_decrypted_blob.bin").write_bytes(clear)

print("[+] Saved: rsa2_decrypted_blob.bin")

if clear.startswith(b"RSA2"):
    print("[+] OK: decrypted data starts with RSA2")
else:
    print("[-] Decrypted data does not start with RSA2")
```
